cmake_minimum_required(VERSION 3.25)
project(QvacParakeetBench LANGUAGES C CXX OBJC OBJCXX)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
    message(FATAL_ERROR "QvacParakeetBench must be cross-compiled for iOS")
endif()
if(NOT IOS_DEVELOPMENT_TEAM)
    message(FATAL_ERROR "IOS_DEVELOPMENT_TEAM is required for device signing")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_METAL ON CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_STATIC ON CACHE BOOL "" FORCE)
set(PARAKEET_BUILD_EXECUTABLES OFF CACHE BOOL "" FORCE)
set(PARAKEET_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(PARAKEET_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(PARAKEET_INSTALL OFF CACHE BOOL "" FORCE)
set(PARAKEET_OPENMP OFF CACHE BOOL "" FORCE)

get_filename_component(PARAKEET_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(IOS_COMMON_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../common")
add_subdirectory("${PARAKEET_SOURCE_DIR}" parakeet)

set(IOS_PRODUCT_NAME QvacParakeetBench)
set(IOS_BUNDLE_IDENTIFIER com.ishanvohra.parakeetbench.qvac)
configure_file(
    "${IOS_COMMON_DIR}/Info.plist.in"
    "${CMAKE_CURRENT_BINARY_DIR}/Info.plist"
    @ONLY)

add_executable(QvacParakeetBench MACOSX_BUNDLE
    "${IOS_COMMON_DIR}/BenchApp.mm"
    QvacAdapter.cpp)
target_include_directories(QvacParakeetBench PRIVATE "${IOS_COMMON_DIR}")
target_link_libraries(QvacParakeetBench PRIVATE
    qvac-parakeet
    "-framework Foundation"
    "-framework UIKit")
set_target_properties(QvacParakeetBench PROPERTIES
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_BINARY_DIR}/Info.plist"
    XCODE_GENERATE_SCHEME TRUE
    XCODE_ATTRIBUTE_CODE_SIGN_STYLE Automatic
    XCODE_ATTRIBUTE_DEVELOPMENT_TEAM "${IOS_DEVELOPMENT_TEAM}"
    XCODE_ATTRIBUTE_IPHONEOS_DEPLOYMENT_TARGET "16.4"
    XCODE_ATTRIBUTE_ONLY_ACTIVE_ARCH YES
    XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER "${IOS_BUNDLE_IDENTIFIER}"
    XCODE_ATTRIBUTE_SUPPORTED_PLATFORMS iphoneos
    XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY "1")
